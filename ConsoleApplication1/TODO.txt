Design a small RAII wrapper for GPU vectors. example:

#pragma once
#include <cuda_runtime.h>
#include <stdexcept>

class GpuVector {
public:
    GpuVector(size_t size)
        : size_(size), data_(nullptr) {
        cudaError_t err = cudaMalloc(&data_, size_ * sizeof(double));
        if (err != cudaSuccess) {
            throw std::runtime_error("cudaMalloc failed: " + std::string(cudaGetErrorString(err)));
        }
    }

    ~GpuVector() {
        if (data_) {
            cudaFree(data_);
        }
    }

    // Disable copy semantics
    GpuVector(const GpuVector&) = delete;
    GpuVector& operator=(const GpuVector&) = delete;

    // Enable move semantics
    GpuVector(GpuVector&& other) noexcept
        : size_(other.size_), data_(other.data_) {
        other.data_ = nullptr;
        other.size_ = 0;
    }

    GpuVector& operator=(GpuVector&& other) noexcept {
        if (this != &other) {
            if (data_) cudaFree(data_);
            data_ = other.data_;
            size_ = other.size_;
            other.data_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }

    double* data() const { return data_; }
    size_t size() const { return size_; }

private:
    size_t size_;
    double* data_;
};

useage:
GpuVector input_d(a.size());
cudaMemcpy(input_d.data(), a.data(), a.size() * sizeof(double), cudaMemcpyHostToDevice);

// Pass input_d.data() to your layers
